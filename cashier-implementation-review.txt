<?php

use App\Http\Controllers\AttendanceController;
use App\Http\Controllers\DashboardController;
use App\Http\Controllers\MemberController;
use App\Http\Controllers\MembershipController;
use App\Http\Controllers\MembershipPlanController;
use App\Http\Controllers\ModulePageController;
use App\Http\Controllers\PaymentController;
use App\Http\Controllers\ProfileController;
use App\Http\Controllers\ReportController;
use App\Http\Controllers\StaffAccountController;
use App\Models\User;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Hash;
use Illuminate\Support\Facades\Route;

Route::get('/', function () {
    return redirect()->route('dashboard');
});

Route::middleware('auth')->group(function () {

    // Dashboard
    Route::get('/dashboard', [DashboardController::class, 'index'])
        ->name('dashboard');

    // Members
    Route::get('/members', [MemberController::class, 'index'])
        ->name('members.index');

    Route::get('/members/create', [MemberController::class, 'create'])
        ->name('members.create');

    Route::post('/members', [MemberController::class, 'store'])
        ->name('members.store');

    Route::get('/members/{member}/edit', [MemberController::class, 'edit'])
        ->name('members.edit');

    Route::put('/members/{member}', [MemberController::class, 'update'])
        ->name('members.update');

    // Membership Plans
    Route::get('/membership-plans', [MembershipPlanController::class, 'index'])
        ->name('membership-plans.index');

    Route::get('/membership-plans/create', [MembershipPlanController::class, 'create'])
        ->name('membership-plans.create');

    Route::post('/membership-plans', [MembershipPlanController::class, 'store'])
        ->name('membership-plans.store');

    Route::get('/membership-plans/{membershipPlan}/edit', [MembershipPlanController::class, 'edit'])
        ->name('membership-plans.edit');

    Route::put('/membership-plans/{membershipPlan}', [MembershipPlanController::class, 'update'])
        ->name('membership-plans.update');

    // Membership Registration and Renewal
    Route::get('/memberships', [MembershipController::class, 'index'])
        ->name('memberships.index');

    Route::get('/memberships/create', [MembershipController::class, 'create'])
        ->name('memberships.create');

    Route::post('/memberships', [MembershipController::class, 'store'])
        ->name('memberships.store');

    Route::post('/memberships/{membership}/renew', [MembershipController::class, 'renew'])
        ->name('memberships.renew');

    // Attendance
    Route::get('/attendance', [AttendanceController::class, 'index'])
        ->name('attendance.index');

    Route::post('/attendance/check-in', [AttendanceController::class, 'checkIn'])
        ->name('attendance.check-in');

    Route::post('/attendance/{attendance}/check-out', [AttendanceController::class, 'checkOut'])
        ->name('attendance.check-out');

    // Payments
    Route::get('/payments', [PaymentController::class, 'index'])
        ->name('payments.index');

    Route::get('/payments/create', [PaymentController::class, 'create'])
        ->name('payments.create');

    Route::post('/payments', [PaymentController::class, 'store'])
        ->name('payments.store');

    // Reports
    Route::get('/reports', [ReportController::class, 'index'])
        ->name('reports.index');

    Route::get('/reports/export', [ReportController::class, 'export'])
        ->name('reports.export');

    // Profile
    Route::get('/profile', [ProfileController::class, 'edit'])
        ->name('profile.edit');

    Route::patch('/profile', [ProfileController::class, 'update'])
        ->name('profile.update');

    Route::delete('/profile', [ProfileController::class, 'destroy'])
        ->name('profile.destroy');

    /*
    |--------------------------------------------------------------------------
    | Super Admin: Staff Management
    |--------------------------------------------------------------------------
    */
    Route::middleware('role:super_admin')
        ->prefix('staff')
        ->name('staff.')
        ->group(function () {
            Route::get('/', [StaffAccountController::class, 'index'])
                ->name('index');

            Route::post('/', [StaffAccountController::class, 'store'])
                ->name('store');

            Route::patch('/{user}/role', [StaffAccountController::class, 'updateRole'])
                ->name('role.update');

            Route::patch('/{user}/reset-password', [StaffAccountController::class, 'resetPassword'])
                ->name('reset-password');

            Route::delete('/{user}', [StaffAccountController::class, 'destroy'])
                ->name('destroy');
        });

    /*
    |--------------------------------------------------------------------------
    | Admin and Super Admin: Reset Cashier/Receptionist Password
    |--------------------------------------------------------------------------
    | These routes never show the existing password. They set a new one.
    */
    Route::middleware('role:admin,super_admin')
        ->prefix('cashier-passwords')
        ->name('cashier-passwords.')
        ->group(function () {
            Route::get('/', function () {
                $users = User::where('role', 'receptionist')
                    ->orderBy('name')
                    ->paginate(15);

                return view('staff.reset-cashier-password', compact('users'));
            })->name('index');

            Route::patch('/{user}', function (Request $request, User $user) {
                abort_unless($user->role === 'receptionist', 404);

                $validated = $request->validate([
                    'password' => ['required', 'string', 'min:12', 'confirmed'],
                ]);

                $user->update([
                    'password' => Hash::make($validated['password']),
                ]);

                return redirect()->route('cashier-passwords.index')
                    ->with('success', 'Cashier password reset successfully.');
            })->name('update');
        });
});

namespace App\Http\Controllers;

use App\Models\Membership;
use App\Models\Payment;
use Illuminate\Http\Request;
use Illuminate\Validation\Rule;

class PaymentController extends Controller
{
    public function index()
    {
        $user = request()->user();

        $paymentsQuery = Payment::with([
            'membership.member',
            'membership.plan',
            'processedBy',
        ]);

        // Cashiers can see only transactions they recorded.
        if ($user->role === 'cashier') {
            $paymentsQuery->where('processed_by', $user->id);
        }

        $payments = $paymentsQuery
            ->orderByDesc('payment_date')
            ->orderByDesc('id')
            ->paginate(15);

        $totalsQuery = Payment::query();

        if ($user->role === 'cashier') {
            $totalsQuery->where('processed_by', $user->id);
        }

        $totalCollected = (clone $totalsQuery)
            ->where('status', 'paid')
            ->sum('amount');

        $todayCollected = (clone $totalsQuery)
            ->where('status', 'paid')
            ->whereDate('payment_date', today())
            ->sum('amount');

        $pendingAmount = (clone $totalsQuery)
            ->where('status', 'pending')
            ->sum('amount');

        return view('payments.index', compact(
            'payments',
            'totalCollected',
            'todayCollected',
            'pendingAmount'
        ));
    }

    public function create()
    {
        $memberships = Membership::with(['member', 'plan'])
            ->orderByDesc('id')
            ->get();

        return view('payments.create', compact('memberships'));
    }

    public function store(Request $request)
    {
        $validated = $request->validate([
            'membership_id' => [
                'required',
                Rule::exists('memberships', 'id'),
            ],
            'amount' => [
                'required',
                'numeric',
                'gt:0',
                'max:99999999.99',
            ],
            'payment_date' => [
                'required',
                'date',
                'before_or_equal:today',
            ],
            'payment_method' => [
                'required',
                Rule::in([
                    'cash',
                    'gcash',
                    'bank_transfer',
                    'card',
                    'other',
                ]),
            ],
            'reference_number' => [
                'nullable',
                'string',
                'max:100',
            ],
            'status' => [
                'required',
                Rule::in(['paid', 'pending', 'failed']),
            ],
            'notes' => [
                'nullable',
                'string',
                'max:2000',
            ],
        ]);

        // Never accept processed_by from the browser form.
        // Attribute the transaction to the authenticated user.
        $validated['processed_by'] = $request->user()->id;

        Payment::create($validated);

        return redirect()
            ->route('payments.index')
            ->with('success', 'Payment record saved successfully.');
    }
}

<?php

namespace App\Http\Controllers;

use App\Models\User;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Hash;
use Illuminate\Validation\Rule;

class StaffAccountController extends Controller
{
    public function index()
    {
        $users = User::orderBy('name')->paginate(15);

        return view('staff.index', compact('users'));
    }

    public function store(Request $request)
    {
        $validated = $request->validate([
            'name' => ['required', 'string', 'max:255'],
            'email' => ['required', 'email', 'max:255', 'unique:users,email'],
            'role' => ['required', Rule::in(['admin', 'receptionist'])],
            'password' => ['required', 'string', 'min:12', 'confirmed'],
        ]);

        User::create([
            'name' => $validated['name'],
            'email' => $validated['email'],
            'role' => $validated['role'],
            'password' => Hash::make($validated['password']),
        ]);

        return redirect()->route('staff.index')
            ->with('success', 'Staff account created successfully.');
    }

    public function updateRole(Request $request, User $user)
    {
        if ($user->role === 'super_admin') {
            return back()->with('error', 'Super Admin accounts cannot be changed here.');
        }

        $validated = $request->validate([
            'role' => ['required', Rule::in(['admin', 'receptionist'])],
        ]);

        $user->update(['role' => $validated['role']]);

        return redirect()->route('staff.index')
            ->with('success', 'Staff role updated successfully.');
    }

    public function destroy(Request $request, User $user)
    {
        if ($user->role === 'super_admin') {
            return back()->with('error', 'Super Admin accounts cannot be deleted here.');
        }

        if ($request->user()->is($user)) {
            return back()->with('error', 'You cannot delete your own account.');
        }

        $user->delete();

        return redirect()->route('staff.index')
            ->with('success', 'Staff account deleted successfully.');
    }

    public function resetPassword(Request $request, User $user)
    {
        if ($user->role === 'super_admin') {
            return back()->with('error', 'This password cannot be reset through staff management.');
        }

        $validated = $request->validate([
            'password' => ['required', 'string', 'min:12', 'confirmed'],
        ]);

        $user->update([
            'password' => Hash::make($validated['password']),
        ]);

        return back()->with('success', 'Password reset successfully.');
    }
}

<?php

namespace App\Http\Controllers;

use App\Models\Attendance;
use App\Models\Member;
use App\Models\Membership;
use App\Models\Payment;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Response;
use Illuminate\Validation\ValidationException;

class ReportController extends Controller
{
    private function filters(Request $request): array
    {
        $validated = $request->validate([
            'type' => ['nullable', 'in:attendance,memberships,payments'],
            'from' => ['nullable', 'date'],
            'to' => ['nullable', 'date', 'after_or_equal:from'],
        ]);

        return [
            'type' => $validated['type'] ?? 'attendance',
            'from' => $validated['from'] ?? now()->startOfMonth()->toDateString(),
            'to' => $validated['to'] ?? today()->toDateString(),
        ];
    }

    public function index(Request $request)
    {
        $filters = $this->filters($request);
        $type = $filters['type'];
        $from = $filters['from'];
        $to = $filters['to'];

        $totalMembers = Member::count();

        $activeMemberships = Membership::whereDate('start_date', '<=', today())
            ->whereDate('end_date', '>=', today())
            ->count();

        $attendanceCount = Attendance::whereBetween('attendance_date', [$from, $to])
            ->count();

        $revenue = Payment::where('status', 'paid')
            ->whereBetween('payment_date', [$from, $to])
            ->sum('amount');

        if ($type === 'memberships') {
            $records = Membership::with(['member', 'plan'])
                ->whereBetween('start_date', [$from, $to])
                ->orderByDesc('start_date')
                ->paginate(20)
                ->withQueryString();
        } elseif ($type === 'payments') {
            $records = Payment::with(['membership.member', 'membership.plan'])
                ->whereBetween('payment_date', [$from, $to])
                ->orderByDesc('payment_date')
                ->orderByDesc('id')
                ->paginate(20)
                ->withQueryString();
        } else {
            $records = Attendance::with('member')
                ->whereBetween('attendance_date', [$from, $to])
                ->orderByDesc('attendance_date')
                ->orderByDesc('check_in_at')
                ->paginate(20)
                ->withQueryString();
        }

        return view('reports.index', compact(
            'filters',
            'type',
            'from',
            'to',
            'records',
            'totalMembers',
            'activeMemberships',
            'attendanceCount',
            'revenue'
        ));
    }

    public function export(Request $request)
    {
        $filters = $this->filters($request);
        $type = $filters['type'];
        $from = $filters['from'];
        $to = $filters['to'];

        $filename = $type . '-report-' . $from . '-to-' . $to . '.csv';

        return Response::streamDownload(function () use ($type, $from, $to) {
            $output = fopen('php://output', 'w');

            // Help spreadsheet programs display UTF-8 correctly.
            fwrite($output, "\xEF\xBB\xBF");

            if ($type === 'memberships') {
                fputcsv($output, [
                    'Membership ID', 'Member Number', 'Member Name',
                    'Plan', 'Start Date', 'End Date', 'Price', 'Status',
                ]);

                Membership::with(['member', 'plan'])
                    ->whereBetween('start_date', [$from, $to])
                    ->orderBy('start_date')
                    ->chunk(200, function ($rows) use ($output) {
                        foreach ($rows as $row) {
                            $status = $row->status === 'cancelled'
                                ? 'Cancelled'
                                : ($row->start_date->isFuture()
                                    ? 'Upcoming'
                                    : ($row->end_date->lt(today())
                                        ? 'Expired'
                                        : 'Active'));

                            fputcsv($output, [
                                $row->id,
                                $row->member?->member_number,
                                $row->member?->full_name,
                                $row->plan?->name,
                                $row->start_date?->toDateString(),
                                $row->end_date?->toDateString(),
                                $row->price,
                                $status,
                            ]);
                        }
                    });
            } elseif ($type === 'payments') {
                fputcsv($output, [
                    'Payment ID', 'Payment Date', 'Member',
                    'Plan', 'Method', 'Reference', 'Amount', 'Status',
                ]);

                Payment::with(['membership.member', 'membership.plan'])
                    ->whereBetween('payment_date', [$from, $to])
                    ->orderBy('payment_date')
                    ->chunk(200, function ($rows) use ($output) {
                        foreach ($rows as $row) {
                            fputcsv($output, [
                                $row->id,
                                $row->payment_date?->toDateString(),
                                $row->membership?->member?->full_name,
                                $row->membership?->plan?->name,
                                $row->payment_method,
                                $row->reference_number,
                                $row->amount,
                                $row->status,
                            ]);
                        }
                    });
            } else {
                fputcsv($output, [
                    'Attendance ID', 'Date', 'Member Number',
                    'Member Name', 'Check In', 'Check Out', 'Method',
                ]);

                Attendance::with('member')
                    ->whereBetween('attendance_date', [$from, $to])
                    ->orderBy('attendance_date')
                    ->chunk(200, function ($rows) use ($output) {
                        foreach ($rows as $row) {
                            fputcsv($output, [
                                $row->id,
                                $row->attendance_date?->toDateString(),
                                $row->member?->member_number,
                                $row->member?->full_name,
                                $row->check_in_at?->format('Y-m-d H:i:s'),
                                $row->check_out_at?->format('Y-m-d H:i:s'),
                                $row->check_in_method,
                            ]);
                        }
                    });
            }

            fclose($output);
        }, $filename, [
            'Content-Type' => 'text/csv; charset=UTF-8',
        ]);
    }
}

class EnsureUserHasRole
{
    public function handle(
        Request $request,
        Closure $next,
        string ...$roles
    ): Response {
        $user = $request->user();

        if (!$user) {
            abort(403, 'You must be signed in.');
        }

        if (!in_array($user->role, $roles, true)) {
            abort(403, 'You are not authorized to access this page.');
        }

        return $next($request);
    }
}
